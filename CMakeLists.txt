cmake_minimum_required(VERSION 3.12)
project(embedded_companion LANGUAGES CXX)
enable_testing()
add_executable(embedded_tests cpp/tests.cpp)
target_include_directories(embedded_tests PRIVATE cpp)
set_target_properties(embedded_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(embedded_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME embedded_tests COMMAND embedded_tests)

add_executable(board_safety_tests cpp/board_safety_tests.cpp)
target_include_directories(board_safety_tests PRIVATE cpp)
set_target_properties(board_safety_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(board_safety_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME board_safety_tests COMMAND board_safety_tests)
